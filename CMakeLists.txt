cmake_minimum_required(VERSION 3.21)
project(LuminousReef VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# A known, pinned release, not a moving main branch. First configure needs Internet.
# To build offline, extract the official GLFW 3.4 source archive into external/glfw.
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(UNIX AND NOT APPLE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/glfw/CMakeLists.txt")
    set(REEF_GLFW_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/external/glfw")
    add_subdirectory(external/glfw EXCLUDE_FROM_ALL)
else()
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.zip)
    FetchContent_MakeAvailable(glfw)
    set(REEF_GLFW_SOURCE "${glfw_SOURCE_DIR}")
endif()
if(NOT EXISTS "${REEF_GLFW_SOURCE}/deps/glad/gl.h")
    message(FATAL_ERROR "Use the complete GLFW 3.4 SOURCE archive; it includes deps/glad/gl.h.")
endif()
add_library(reef_glad STATIC src/glad_impl.c)
target_include_directories(reef_glad PUBLIC "${REEF_GLFW_SOURCE}/deps")

add_executable(LuminousReef
    src/main.cpp src/Graphics.cpp src/Geometry.cpp src/Scene.cpp src/UI.cpp
    src/Math.hpp src/Procedural.hpp src/Graphics.hpp src/Geometry.hpp src/Scene.hpp src/UI.hpp)
target_include_directories(LuminousReef PRIVATE src)
target_link_libraries(LuminousReef PRIVATE glfw reef_glad)
target_compile_definitions(LuminousReef PRIVATE REEF_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(LuminousReef PRIVATE /W4 /permissive- /utf-8)
    target_compile_definitions(LuminousReef PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(LuminousReef PRIVATE -Wall -Wextra -Wpedantic)
endif()
# Shaders stay beside the executable, regardless of Debug/Release working directory.
add_custom_target(reef_shaders ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:LuminousReef>/shaders")
add_dependencies(LuminousReef reef_shaders)
set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT LuminousReef)
set_property(TARGET LuminousReef PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:LuminousReef>")

include(CTest)
if(BUILD_TESTING)
    add_executable(reef_math_tests tests/math_tests.cpp)
    target_include_directories(reef_math_tests PRIVATE src)
    add_test(NAME MathAndProceduralTests COMMAND reef_math_tests)
endif()
